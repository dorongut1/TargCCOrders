CREATE PROCEDURE [dbo].[c__SystemAuditAdd] 
  (
   @TableName varchar(50) 
  ,@RowID bigint 
  ,@Operation varchar(10) 
  ,@OccurredAt datetime 
  --,@SqlCurrentUser nvarchar(50) 
  ,@ChangedByUser nvarchar(50) 
  ,@ActiveLoginID bigint 
  --,@SqlSystemUser nvarchar(50) 
  --,@SqlAppName nvarchar(250) 
  --,@SqlHostName varchar(50) 
  ,@Changes nvarchar(max) 
  )
AS  
  SET NOCOUNT ON

  DECLARE @ID bigint

  INSERT INTO [dbo].[c_SystemAudit]
    ([TableName]
    ,[RowID]
    ,[Operation]
    ,[OccurredAt]
    ,[SqlCurrentUser]
    ,[ChangedByUser]
    ,[ActiveLoginID]
    ,[SqlSystemUser]
    ,[SqlAppName]
    ,[SqlHostName]
    ,[Changes]
    )
  VALUES 
    (@TableName
    ,@RowID
    ,@Operation
    ,@OccurredAt
    ,Current_User
    ,@ChangedByUser
    ,@ActiveLoginID
    ,System_User
    ,APP_NAME()
    ,HOST_NAME()
    ,@Changes
    )
  SET @ID = SCOPE_IDENTITY()
  
  SELECT @ID AS ID
  
  RETURN

GO
